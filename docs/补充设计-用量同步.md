# 用量同步功能补充设计

## 1. 需求理解

### 1.1 核心场景

- 供应商：火山方舟、Moonshot、百度千帆、腾讯混元等
- 用量限制：5小时用量限制、周用量限制
- 需求：实时查询每个套餐的当前用量，精确掌握配额消耗

### 1.2 用量类型

| 用量类型 | 说明 | 重置周期 |
|---------|------|---------|
| 5小时用量 | 最近5小时内的累计用量 | 每5小时重置 |
| 周用量 | 本周内的累计用量 | 每周重置 |

### 3.2 火山方舟 (ByteDance) 详细设计

```python
# 火山方舟用量查询示例
class VolcanoProviderAdapter(BaseProviderAdapter):
    """火山方舟用量查询适配器"""
    
    BASE_URL = "https://ark.cn-beijing.volces.com"
    
    async def get_usage(self, provider: Provider) -> dict:
        """获取用量"""
        headers = {
            "Authorization": f"Bearer {provider.api_key}",
            "Content-Type": "application/json"
        }
        
        # 调用用量查询API
        response = await self.client.get(
            f"{self.BASE_URL}/api/v3/quote/usage",
            headers=headers,
            params={"project_id": provider.config.get("project_id")}
        )
        
        data = response.json()
        
        # 解析响应
        return {
            "hourly": {
                "used": data.get("hourly_quota", {}).get("used", 0),
                "limit": data.get("hourly_quota", {}).get("limit", 0)
            },
            "weekly": {
                "used": data.get("daily_quota", {}).get("used", 0),
                "limit": data.get("daily_quota", {}).get("limit", 0)
            }
        }
```

## 4. 通用适配器架构

### 4.1 适配器接口定义

```python
from abc import ABC, abstractmethod
from typing import Dict, Optional
from dataclasses import dataclass

@dataclass
class QuotaUsage:
    """配额使用情况"""
    used: int
    limit: int
    percent: float
    reset_at: Optional[str] = None

@dataclass
class ProviderQuota:
    """供应商配额"""
    hourly: QuotaUsage
    weekly: QuotaUsage
    synced_at: str

class BaseProviderAdapter(ABC):
    """供应商适配器基类"""
    
    @abstractmethod
    async def get_usage(self, provider: Provider) -> ProviderQuota:
        """获取用量"""
        pass
    
    @abstractmethod
    async def test_connection(self, provider: Provider) -> bool:
        """测试连接"""
        pass
```

### 4.2 适配器注册中心

```python
class ProviderAdapterRegistry:
    """适配器注册中心"""
    
    _adapters: Dict[str, Type[BaseProviderAdapter]] = {}
    
    @classmethod
    def register(cls, provider_type: str):
        """注册适配器"""
        def decorator(adapter_class: Type[BaseProviderAdapter]):
            cls._adapters[provider_type] = adapter_class
            return adapter_class
        return decorator
    
    @classmethod
    def get_adapter(cls, provider_type: str) -> BaseProviderAdapter:
        """获取适配器"""
        adapter_class = cls._adapters.get(provider_type)
        if not adapter_class:
            raise ValueError(f"Unknown provider type: {provider_type}")
        return adapter_class()

# 注册各供应商适配器
@ProviderAdapterRegistry.register("volcengine")
class VolcanoAdapter(BaseProviderAdapter):
    pass

@ProviderAdapterRegistry.register("moonshot")
class MoonshotAdapter(BaseProviderAdapter):
    pass

@ProviderAdapterRegistry.register("baidu")
class BaiduAdapter(BaseProviderAdapter):
    pass
```

### 7.2 同步任务代码

```python
class QuotaSyncService:
    """用量同步服务"""
    
    async def sync_provider_quota(self, provider_id: str):
        """同步单个供应商配额"""
        provider = await self.get_provider(provider_id)
        adapter = ProviderAdapterRegistry.get_adapter(provider.type)
        
        try:
            # 调用适配器获取用量
            quota = await adapter.get_usage(provider)
            
            # 更新数据库
            await self.update_provider_quota(provider_id, quota)
            
            # 检查告警
            await self.check_alerts(provider, quota)
            
        except Exception as e:
            logger.error(f"同步失败: {provider_id}, error: {e}")
            await self.record_sync_error(provider_id, str(e))
    
    async def sync_all_providers(self):
        """同步所有启用的供应商"""
        providers = await self.get_enabled_providers()
        
        tasks = [
            self.sync_provider_quota(p.provider_id) 
            for p in providers
        ]
        
        await asyncio.gather(*tasks, return_exceptions=True)
```

### 7.3 告警规则
